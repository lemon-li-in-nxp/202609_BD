function gridToRects(g){
    const CW=100/g[0].length,CH=100/g.length,out=[];
    for(let y=0;y<g.length;y++){
        const row=g[y];let x=0;
        while(x<row.length){
            if(row[x]==="1"){
                let x2=x;
                while(x2<row.length&&row[x2]==="1")x2++;
                out.push([x*CW,y*CH,(x2-x)*CW,CH]);
                x=x2;
            }else{x++;}
        }
    }
    return out;
}

function pointBlockedOn(mapId,px,py){
    const coll=MAPS[mapId].coll||[];
    for(const c of coll){
        if(px>=c[0]&&px<=c[0]+c[2]&&py>=c[1]&&py<=c[1]+c[3])return true;

}
function sanitizeChests(){
    for(const key in MAPS){
        (MAPS[key].chests||[]).forEach(function(c){
            if(!pointBlockedOn(key,c.x,c.y))return;
            let best=null,bd=-1;
            for(let r=0;r<24;r++){
                for(let col=0;col<20;col++){
                    const px=col*5+2.5,py=r*(100/24)+100/48;
                    if(pointBlockedOn(key,px,py))continue;
                    const d=Math.pow(px-c.x,2)+Math.pow(py-c.y,2);
                    if(bd<0||d<bd){bd=d;best=[px,py];}
                }
            }
            if(best){c.x=best[0];c.y=Math.round(best[1]*1000)/1000;}
        });
    }
}

function measureContentTop(canvas){
    if(!canvas)return 0;
    const ctx=canvas.getContext("2d");
    let data;
    try{data=ctx.getImageData(0,0,64,64).data;}catch(e){return 0;}
    for(let y=0;y<64;y++){
        for(let x=0;x<64;x++){
            if(data[(y*64+x)*4+3]>15)return y;
        }
    }
    return 0;
}

const _imgCache={};
function cachedImg(src){
    if(!_imgCache[src]){const im=new Image();im.src=src;_imgCache[src]=im;}
    return _imgCache[src];
}
function drawCharacter(canvasId){
    const canvas=document.getElementById(canvasId);
    if(!canvas)return;
    const ctx=canvas.getContext("2d");
    const body=cachedImg(bodyFiles[bodyIndex]);
    const hair=cachedImg(hairFiles[hairIndex]);
    const cloth=cachedImg(clothesFiles[clothesIndex]);
    if(!(body.complete&&body.naturalWidth>0)||!(hair.complete&&hair.naturalWidth>0)||!(cloth.complete&&cloth.naturalWidth>0)){
        [body,hair,cloth].forEach(function(im){
            if(!im._q){im._q=true;im.onload=function(){im._q=false;drawCharacter(canvasId);};}
        });
        return;
    }
    ctx.clearRect(0,0,canvas.width,canvas.height);
    const size=64;
    const moving=canvasId==="gameCanvas"&&isMoving;
    let frameX=moving?currentFrame*64:STAND_X*64;
    let frameY=moving?direction*64:STAND_Y*64;
    const mask=(canvasId==="characterCanvas")?charSel:((P&&P.layers)||{body:true,hair:true,clothes:true,wings:true});
    if(mask.wings&&wingFiles[wingIndex]){
        const wing=cachedImg(wingFiles[wingIndex]);
        if(wing.complete&&wing.naturalWidth>0)ctx.drawImage(wing,frameX,frameY,size,size,0,0,size,size);
    }
    if(mask.body)ctx.drawImage(body,frameX,frameY,size,size,0,0,size,size);
    if(mask.clothes)ctx.drawImage(cloth,frameX,frameY,size,size,0,0,size,size);
    if(mask.hair)ctx.drawImage(hair,frameX,frameY,size,size,0,0,size,size);
    if(canvasId==="gameCanvas"){
        const nameEl=document.getElementById("playerNameGame");
        if(nameEl)nameEl.style.bottom=labelBottomPx(48,measureContentTop(canvas))+"px";
    }
}

let view={s:1,x0:0,y0:0};
function computeView(){
    const area=document.getElementById("gameArea");
    if(!area)return;
    const sx=area.clientWidth||1, sy=area.clientHeight||1;
    const s=Math.max(sx/384,sy/480);
    const sw=384*s, sh=480*s;
    view={s:s,x0:(sw-sx)/2,y0:(sh-sy)/2,sx:sx,sy:sy};
}
function toStageFx(ppx){return (ppx/100*384*view.s-view.x0)/view.sx*100;}
function toStageFy(ppy){return (ppy/100*480*view.s-view.y0)/view.sy*100;}
function toImageX(ppx){return (ppx/100*view.sx+view.x0)/(384*view.s)*100;}
function toImageY(ppy){return (ppy/100*view.sy+view.y0)/(480*view.s)*100;}
function refreshView(){
    computeView();

    if(p){p.style.left=toStageFx(playerX)+"%";p.style.top=toStageFy(playerY)+"%";}

const npcImgCache={};
const npcHeadTop={};
function drawNpcOn(id,canvas){
    const n=NPCS[id];
    if(!canvas)return;
    const ctx=canvas.getContext("2d");
    const applyOffset=function(){
        const top=measureContentTop(canvas);
        const dispH=npcDisplayH(id);
        npcHeadTop[id]=top;
        const info=canvas.parentElement?canvas.parentElement.querySelector(".npcInfo"):null;
        if(info)info.style.bottom=labelBottomPx(dispH,top)+"px";
    };
    const render=function(img){
        ctx.clearRect(0,0,64,64);
        const sc=Math.min(64/img.naturalWidth,64/img.naturalHeight);
        const w=Math.round(img.naturalWidth*sc),h=Math.round(img.naturalHeight*sc);
        ctx.drawImage(img,0,0,img.naturalWidth,img.naturalHeight,(64-w)/2,(64-h)/2,w,h);
        applyOffset();
    };
    if(npcImgCache[id]){
        const cimg=npcImgCache[id];
        if(cimg.complete)render(cimg);
        else cimg.onload=function(){render(cimg);};
        return;
    }
    const im=new Image();im.src=n.img;
    npcImgCache[id]=im;
    im.onload=function(){render(im);};
}

function isBlocked(x,y){
    for(const c of collisions){
        if(x>=c[0]&&x<=c[0]+c[2]&&y>=c[1]&&y<=c[1]+c[3])return true;
    }
    return false;
}
function onAreaClick(e){
    if(isDialogOpen||anyOverlayOpen())return;
    const area=document.getElementById("gameArea");
    const rect=area.getBoundingClientRect();
    let ppx=((e.touches?(e.touches[0].clientX):e.clientX)-rect.left)/rect.width*100;
    let ppy=((e.touches?(e.touches[0].clientY):e.clientY)-rect.top)/rect.height*100;
    let tx=toImageX(ppx),ty=toImageY(ppy);
    tx=Math.min(98,Math.max(2,tx));
    ty=Math.min(97,Math.max(3,ty));
    if(Math.sqrt(Math.pow(tx-playerX,2)+Math.pow(ty-playerY,2))<1.5)return;
    if(isBlocked(tx,ty)){toast("被擋住了！");return;}
    targetX=tx;targetY=ty;isMoving=true;
}

function stopMovement(){
    isMoving=false;currentFrame=0;direction=40;
    drawCharacter("gameCanvas");
}
const MOVE_SPEED=35;
let lastFrame=performance.now();
function updateLoop(ts){
    requestAnimationFrame(updateLoop);
    const dt=Math.min(0.05,(ts-lastFrame)/1000||0);
    lastFrame=ts;
    if(!isMoving||isDialogOpen||anyOverlayOpen())return;
    const dx=targetX-playerX,dy=targetY-playerY;
    const dist=Math.sqrt(dx*dx+dy*dy);
    if(dist<1){
        stopMovement();
        updateZoneBtn();
        return;
    }
    const step=Math.min(MOVE_SPEED*dt,dist);
    const nx=playerX+dx/dist*step,ny=playerY+dy/dist*step;
    if(isBlocked(nx,ny)){
        if(!isBlocked(nx,playerY)){playerX=nx;}
        else if(!isBlocked(playerX,ny)){playerY=ny;}
        else{stopMovement();updateZoneBtn();return;}
    }else{playerX=nx;playerY=ny;}
    direction=Math.abs(dx)>Math.abs(dy)?(dx>0?41:39):(dy>0?40:38);
    frameTimer+=dt;
    if(frameTimer>=0.12){frameTimer=0;currentFrame=(currentFrame+1)%8;}
    document.getElementById("playerCharacter").style.left=toStageFx(playerX)+"%";
    document.getElementById("playerCharacter").style.top=toStageFy(playerY)+"%";
    drawCharacter("gameCanvas");
    updateZoneBtn();
}
requestAnimationFrame(updateLoop);

function compressImage(src,cb){
    const img=new Image();
    img.onload=function(){
        const MAX=480;
        let w=img.width,h=img.height;
        if(w>MAX){h=Math.round(h*MAX/w);w=MAX;}
        const cv=document.createElement("canvas");
        cv.width=w;cv.height=h;
        const ctx=cv.getContext("2d");
        ctx.drawImage(img,0,0,w,h);
        cb(cv.toDataURL("image/jpeg",0.6));
    };
    img.src=src;
}

let idb=null;
function idbOpen(){
    return new Promise(function(res,rej){
        const r=indexedDB.open("perthquest",1);
        r.onupgradeneeded=function(){r.result.createObjectStore("full",{keyPath:"qid"});};
        r.onsuccess=function(){res(r.result);};
        r.onerror=function(){rej(r.error);};
    });
}
function idbStore(obj){
    idbOpen().then(function(db){
        const tx=db.transaction("full","readwrite");
        tx.objectStore("full").put(obj);
    }).catch(function(){});
}
function idbLoad(){
    return idbOpen().then(function(db){
        return new Promise(function(res){
            const tx=db.transaction("full","readonly");
            const req=tx.objectStore("full").getAll();
            req.onsuccess=function(){res(req.result||[]);};
        });
    }).catch(function(){return[];});
}

function dailySubmitQ(i){

    if(q.type==="num"){
        const el=document.getElementById("dailyNum"+i);
        const n=el?parseInt(el.value,10):NaN;
        if(isNaN(n)||n<0){toast("請先填入數字！");return;}
        v=Math.floor(n);
        reward=v*10;
    }else{
        if(dailyAnswers[q.id]===undefined){toast("請先選擇答案！");return;}
        v=dailyAnswers[q.id];
        const c=q.choices.find(function(x){return x.pts===v;});
        reward=c?c.pts:0;
    }
    P.daily.answers[q.id]=v;
    if(q.id==="poo"){P.stats.poo+=v;}
    if(q.id==="km"){P.stats.km+=v;}
    if(q.id==="kangaroo"){P.stats.kangaroo+=v;}

function useGirlCard(k){
    const g=girlQuiz;
    if(!g)return;
    if(g.done){toast("這題已經答完，不能再出卡片囉");return;}
    if(g.used[k])return;
    if(g.cardUsed){toast("這題已經用過卡片囉");return;}
    g.used[k]=true;g.cardUsed=true;
    if(k==="x2"||k==="x5"){g.mult=k==="x2"?2:5;}
    else if(k==="half"){
        const q=GIRL_QUIZ[g.i];
        const wrongs=[];
        for(let i=0;i<q.o.length;i++){if(i!==q.a)wrongs.push(i);}
        wrongs.sort(function(){return Math.random()-0.5;});
        g.removed={};g.removed[wrongs[0]]=true;g.removed[wrongs[1]]=true;
    }
    else if(k==="call"){
        document.getElementById("girlCallBox").classList.add("show");
    }
    renderGirlQ();
}
function pickerGirl(i){
    const g=girlQuiz;
    if(!g||g.done)return;
    g.done=true;g.pick=i;
    const q=GIRL_QUIZ[g.i];
    const correct=i===q.a;
    g.correct=correct;
    const base=correct?1000:-1000;
    const delta=base*(g.mult||1);
    g.delta=delta;
    g.total+=delta;
    P.points+=delta;
    saveProfile();
    updateHUD();
    renderGirlQ();
}

function rebuildCollisionLayer(){
    const layer=document.getElementById("collisionLayer");
    layer.innerHTML="";
    if(!collisions)return;
    collisions.forEach(function(c,idx){
        const box=document.createElement("div");
        box.style.cssText="position:absolute;left:"+toStageFx(c[0])+"%;top:"+toStageFy(c[1])+"%;width:"+(toStageFx(c[0]+c[2])-toStageFx(c[0]))+"%;height:"+(toStageFy(c[1]+c[3])-toStageFy(c[1]))+"%;background:rgba(255,0,0,.25);border:1px solid #f55;pointer-events:auto;";
        box.dataset.idx=idx;
        box.onclick=function(e){
            e.stopPropagation();
            collisions.splice(idx,1);
            box.remove();
        };
        layer.appendChild(box);
    });
}
const startDraw={x:0,y:0,active:false,box:null};
function editDrawStart(e){
    const rect=document.getElementById("gameArea").getBoundingClientRect();
    startDraw.x=toImageX((e.clientX-rect.left)/rect.width*100);
    startDraw.y=toImageY((e.clientY-rect.top)/rect.height*100);
    startDraw.active=true;
    startDraw.box=document.createElement("div");
    startDraw.box.style.cssText="position:absolute;border:2px dashed #ffe14d;background:rgba(255,225,77,.15);pointer-events:none;";
    document.getElementById("collisionLayer").appendChild(startDraw.box);
}
function editDrawMove(e){
    if(!startDraw.active)return;
    const rect=document.getElementById("gameArea").getBoundingClientRect();
    const cx=toImageX((e.clientX-rect.left)/rect.width*100);
    const cy=toImageY((e.clientY-rect.top)/rect.height*100);
    startDraw.box.style.left=toStageFx(Math.min(startDraw.x,cx))+"%";
    startDraw.box.style.top=toStageFy(Math.min(startDraw.y,cy))+"%";
    startDraw.box.style.width=(toStageFx(Math.max(startDraw.x,cx))-toStageFx(Math.min(startDraw.x,cx)))+"%";
    startDraw.box.style.height=(toStageFy(Math.max(startDraw.y,cy))-toStageFy(Math.min(startDraw.y,cy)))+"%";
}
function editDrawEnd(e){
    if(!startDraw.active)return;
    startDraw.active=false;
    const rect=document.getElementById("gameArea").getBoundingClientRect();
    const cx=toImageX((e.clientX-rect.left)/rect.width*100);
    const cy=toImageY((e.clientY-rect.top)/rect.height*100);
    const w=Math.abs(cx-startDraw.x),h=Math.abs(cy-startDraw.y);
    if(startDraw.box){startDraw.box.remove();startDraw.box=null;}
    if(w<0.5||h<0.5)return;
    collisions.push([Math.min(startDraw.x,cx),Math.min(startDraw.y,cy),w,h]);
    rebuildCollisionLayer();
}

    if(e.key==="F2"){

        editMode=!editMode;

        toast(editMode?"碰撞編輯 ON（點紅框刪除、拖曳新增）":"碰撞編輯 OFF");
    }
    if(e.key==="F3"){
        e.preventDefault();
        const json=JSON.stringify(collisions.map(function(c){return {x:c[0],y:c[1],width:c[2],height:c[3]};}),null,4);

        a.href=URL.createObjectURL(blob);
        a.download=currentMap+".json";
        a.click();
        setTimeout(function(){URL.revokeObjectURL(a.href);},1000);
    }
});
