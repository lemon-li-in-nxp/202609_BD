"images/clothes/clothes5.png"

const wingFiles = [
null,
"images/wings/wings1.png",
"images/wings/wings2.png"
];

let bodyIndex=0;
let hairIndex=0;
let clothesIndex=0;
let wingIndex=0;

let currentUser = null;

let currentFrame = 0;

let direction = 40;

/*
38 = 上
39 = 左
40 = 下
41 = 右
*/

let isMoving = false;
let playerX = 40;
let playerY = 80;

let targetX = 40;
let targetY = 80;
function drawCharacter(canvasId){

    const canvas =
    document.getElementById(canvasId);

    if(!canvas) return;

    const ctx =
    canvas.getContext("2d");

    ctx.clearRect(
        0,0,
        canvas.width,
        canvas.height
    );

    const body =
    new Image();

    const hair =
    new Image();

    const cloth =
    new Image();

    const wing =
    new Image();

    body.src =
    bodyFiles[bodyIndex];

    hair.src =
    hairFiles[hairIndex];

    cloth.src =
    clothesFiles[clothesIndex];

if(wingFiles[wingIndex]){
    wing.src = wingFiles[wingIndex];
}

    Promise.all([

        new Promise(r=>body.onload=r),
        new Promise(r=>hair.onload=r),
        new Promise(r=>cloth.onload=r),
        new Promise(r=>{

    if(wingFiles[wingIndex]){
        wing.onload = r;
    }
    else{
        r();
    }

})

    ]).then(()=>{

const size = 64;

let frameX;
let frameY;

if(
    canvasId === "gameCanvas"
    &&
    isMoving
){

    frameX =
    currentFrame * 64;

}
else{

    frameX = 0;

}

if(canvasId === "gameCanvas"){

    frameY =
    direction * 64;

}
else{

    frameY =
    40 * 64;

}

if(wingFiles[wingIndex]){

    ctx.drawImage(
        wing,
        frameX,frameY,size,size,
        0,0,size,size
    );

}

        ctx.drawImage(
            body,
            frameX,frameY,size,size,
            0,0,size,size
        );

        ctx.drawImage(
            cloth,
            frameX,frameY,size,size,
            0,0,size,size
        );

        ctx.drawImage(
            hair,
            frameX,frameY,size,size,
            0,0,size,size
        );

    });

}

function updateCharacter(){

    drawCharacter("characterCanvas");
    drawCharacter("previewCanvas");
    drawCharacter("gameCanvas");

}

function changeBody(step){

    bodyIndex += step;

    if(bodyIndex < 0)
        bodyIndex =
        bodyFiles.length - 1;

    if(bodyIndex >= bodyFiles.length)
        bodyIndex = 0;

    document.getElementById(
    "bodyText"
    ).innerText =
    "Body " + (bodyIndex+1);

    updateCharacter();
}

if(wingIndex===0){

        username !== "test" ||

        return;
    }

    currentUser = username;

if(characterData){

}

function movePlayer(event){

    const area =
    document.getElementById(
        "gameArea"
    );

    const rect =
    area.getBoundingClientRect();

    targetX =
    (
        (event.clientX - rect.left)
        / rect.width
    ) * 100;

    targetY =
    (
        (event.clientY - rect.top)
        / rect.height
    ) * 100;

    if(targetX < 5){
        targetX = 5;
    }

    if(targetX > 90){
        targetX = 90;
    }

    if(targetY < 10){
        targetY = 10;
    }

    if(targetY > 85){
        targetY = 85;
    }

    isMoving = true;

}

}

    const saveData = {

        body: bodyIndex,
        hair: hairIndex,
        clothes: clothesIndex,

alert(

);

showCharacterPreview();

}
window.onload = function(){

    setInterval(()=>{

        if(isMoving){

            const dx =
            targetX - playerX;

            const dy =
            targetY - playerY;

            const distance =
            Math.sqrt(
                dx*dx +
                dy*dy
            );

            if(distance < 1){

                isMoving = false;

                currentFrame = 0;

            }
            else{

                playerX +=
                dx * 0.08;

                playerY +=
                dy * 0.08;

                if(
                    Math.abs(dx)
                    >
                    Math.abs(dy)
                ){

                    if(dx > 0){

                        direction = 41;

                    }else{

                        direction = 39;

                    }

                }
                else{

                    if(dy > 0){

                        direction = 40;

                    }else{

                        direction = 38;

                    }

                }

                currentFrame++;

                if(currentFrame > 7){

                    currentFrame = 0;

                }

            }

            document.getElementById(
                "playerCharacter"
            ).style.left =
            playerX + "%";

            document.getElementById(
                "playerCharacter"
            ).style.top =
            playerY + "%";

            drawCharacter(
                "gameCanvas"
            );

        }

    },100);

    updateCharacter();

};
